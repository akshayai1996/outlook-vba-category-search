Private Sub ResetOptions()
    Dim i As Long
    For i = 0 To lstCategories.ListCount - 1
        lstCategories.Selected(i) = False
    Next i
    optAnd.Value = True
    cboScope.ListIndex = 2
    chkAttach.Value = False
    txtSender.Text = "": txtKeyword.Text = "": txtFrom.Text = "": txtTo.Text = ""
End Sub

Private Sub LoadCategories()
    Dim cat As Outlook.Category
    Dim names() As String
    Dim n As Long, i As Long, j As Long, tmp As String
    On Error GoTo EH
    n = Application.Session.Categories.Count
    lstCategories.Clear
    If n = 0 Then Exit Sub
    ReDim names(0 To n - 1)
    For Each cat In Application.Session.Categories
        names(i) = cat.Name: i = i + 1
    Next cat
    For i = 1 To n - 1
        tmp = names(i): j = i - 1
        Do While j >= 0
            If StrComp(names(j), tmp, vbTextCompare) <= 0 Then Exit Do
            names(j + 1) = names(j): j = j - 1
        Loop
        names(j + 1) = tmp
    Next i
    For i = 0 To n - 1
        lstCategories.AddItem names(i)
    Next i
    Exit Sub
EH:
    MsgBox "Could not load categories: " & Err.Description, vbExclamation
End Sub

Private Function ReadOptions(ByRef opt As SearchOptions) As Boolean
    Dim i As Long, k As Long
    Dim sel() As String
    For i = 0 To lstCategories.ListCount - 1
        If lstCategories.Selected(i) Then k = k + 1
    Next i
    If k > 0 Then
        ReDim sel(0 To k - 1)
        k = 0
        For i = 0 To lstCategories.ListCount - 1
            If lstCategories.Selected(i) Then sel(k) = lstCategories.List(i): k = k + 1
        Next i
        opt.Cats = sel
    End If
    opt.CatCount = k
    opt.MatchAll = optAnd.Value
    Select Case cboScope.ListIndex
        Case 0: opt.EntireMailbox = False: opt.IncludeSubfolders = False: opt.IncludeShared = False
        Case 1: opt.EntireMailbox = False: opt.IncludeSubfolders = True:  opt.IncludeShared = False
        Case 2: opt.EntireMailbox = True:  opt.IncludeSubfolders = True:  opt.IncludeShared = False
        Case Else: opt.EntireMailbox = True: opt.IncludeSubfolders = True: opt.IncludeShared = True
    End Select
    opt.OnlyAttachments = chkAttach.Value
    opt.Sender = Trim$(txtSender.Text)
    opt.Keyword = Trim$(txtKeyword.Text)
    opt.SearchBody = False
    If Len(Trim$(txtFrom.Text)) > 0 Or Len(Trim$(txtTo.Text)) > 0 Then
        opt.UseDate = True
        opt.DateFrom = DateSerial(1900, 1, 1)
        opt.DateTo = DateSerial(2999, 12, 31)
        If Len(Trim$(txtFrom.Text)) > 0 Then
            If Not IsDate(txtFrom.Text) Then
                MsgBox "Enter the 'from' date as yyyy-mm-dd.", vbExclamation, "Category Search"
                Exit Function
            End If
            opt.DateFrom = CDate(txtFrom.Text)
        End If
        If Len(Trim$(txtTo.Text)) > 0 Then
            If Not IsDate(txtTo.Text) Then
                MsgBox "Enter the 'to' date as yyyy-mm-dd.", vbExclamation, "Category Search"
                Exit Function
            End If
            opt.DateTo = CDate(txtTo.Text)
        End If
        If opt.DateFrom > opt.DateTo Then
            MsgBox "'from' date is after 'to' date.", vbExclamation, "Category Search"
            Exit Function
        End If
    End If
    If opt.CatCount = 0 And Len(opt.Sender) = 0 And Len(opt.Keyword) = 0 _
       And Not opt.OnlyAttachments And Not opt.UseDate Then
        MsgBox "Pick at least one category (or fill an optional filter).", vbInformation, "Category Search"
        Exit Function
    End If
    ReadOptions = True
End Function

Private Sub btnSearch_Click()
    Dim opt As SearchOptions
    Dim n As Long
    On Error GoTo EH
    If Not ReadOptions(opt) Then Exit Sub
    btnSearch.Enabled = False
    Me.MousePointer = fmMousePointerHourGlass
    n = ExecuteSearch(opt)
    Me.MousePointer = fmMousePointerDefault
    btnSearch.Enabled = True
    If n = 0 Then
        MsgBox "No matching emails found.", vbInformation, "Category Search"
    Else
        frmResults.LoadResults
        frmResults.Show vbModeless
    End If
    Exit Sub
EH:
    Me.MousePointer = fmMousePointerDefault
    btnSearch.Enabled = True
    MsgBox "Error: " & Err.Description, vbExclamation, "Category Search"
End Sub

Private Sub btnClear_Click()

    lblStatus.Caption = "Cleared."
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
